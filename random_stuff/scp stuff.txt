import shutil
from pathlib import Path

source_root = Path("mouse-control")
target_root = Path("SHIT")

    if b_dir.is_dir():
        rel = b_dir.relative_to(source_root)
        target = target_root / rel

        shutil.move(str(b_dir), str(target))
